    <script>
        document.addEventListener('DOMContentLoaded', function () {
            // Constants
            const R = 8.31446261815324; // Universal gas constant in J/(mol·K)

            // Get DOM elements
            const tempSlider = document.getElementById('temperature');
            const massSlider = document.getElementById('molarMass');
            const tempValueSpan = document.getElementById('tempValue');
            const massValueSpan = document.getElementById('massValue');
            const vpSpan = document.getElementById('vp');
            const vavgSpan = document.getElementById('vavg');
            const vrmsSpan = document.getElementById('vrms');
            const ctx = document.getElementById('mbChart').getContext('2d');

            // Maxwell-Boltzmann distribution function f(v)
            function maxwellBoltzmann(v, T, M_kg) {
                const a = Math.pow(M_kg / (2 * Math.PI * R * T), 1.5);
                const b = 4 * Math.PI * v * v;
                const c = Math.exp(-(M_kg * v * v) / (2 * R * T));
                return a * b * c;
            }

            // Chart.js instance
            let mbChart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels: [],
                    datasets: [{
                        label: 'Maxwell-Boltzmann Distribution',
                        data: [],
                        borderColor: '#3b82f6',
                        backgroundColor: 'rgba(59, 130, 246, 0.2)',
                        borderWidth: 2,
                        fill: true,
                        pointRadius: 0,
                        tension: 0.4
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    animation: {
                        duration: 300 // Smooth but quick animation
                    },
                    scales: {
                        x: {
                            type: 'linear',
                            title: {
                                display: true,
                                text: 'Speed (m/s)',
                                font: { size: 14, weight: 'bold' },
                                color: '#1f2937'
                            },
                            ticks: {
                                maxTicksLimit: 10,
                                color: '#1f2937'
                            },
                            grid: {
                                color: 'rgba(0, 0, 0, 0.05)'
                            }
                        },
                        y: {
                            title: {
                                display: true,
                                text: 'Probability Density (s/m)',
                                font: { size: 14, weight: 'bold' },
                                color: '#1f2937'
                            },
                            beginAtZero: true,
                            ticks: {
                                callback: function(value) {
                                    return value.toExponential(1);
                                },
                                color: '#1f2937'
                            },
                            grid: {
                                color: 'rgba(0, 0, 0, 0.05)'
                            }
                        }
                    },
                    plugins: {
                        legend: { display: false },
                        tooltip: {
                            enabled: true,
                            mode: 'index',
                            intersect: false,
                            backgroundColor: 'rgba(0, 0, 0, 0.8)',
                            titleFont: { size: 14 },
                            bodyFont: { size: 12 }
                        },
                        annotation: {
                            annotations: {
                                vpLine: {
                                    type: 'line',
                                    xMin: 0,
                                    xMax: 0,
                                    borderColor: '#3b82f6',
                                    borderWidth: 2,
                                    borderDash: [5, 5],
                                    label: {
                                        content: 'v_p',
                                        enabled: true,
                                        position: 'top',
                                        backgroundColor: 'rgba(59 | 130 | 246 | 0.8)',
                                        color: '#ffffff',
                                        font: { size: 12, weight: 'bold' }
                                    }
                                },
                                vavgLine: {
                                    type: 'line',
                                    xMin: 0,
                                    xMax: 0,
                                    borderColor: '#10b981',
                                    borderWidth: 2,
                                    borderDash: [5, 5],
                                    label: {
                                        content: 'v_avg',
                                        enabled: true,
                                        position: 'top',
                                        backgroundColor: 'rgba(16 | 185 | 129 | 0.8)',
                                        color: '#ffffff',
                                        font: { size: 12, weight: 'bold' }
                                    }
                                },
                                vrmsLine: {
                                    type: 'line',
                                    xMin: 0,
                                    xMax: 0,
                                    borderColor: '#ef4444',
                                    borderWidth: 2,
                                    borderDash: [5, 5],
                                    label: {
                                        content: 'v_rms',
                                        enabled: true,
                                        position: 'top',
                                        backgroundColor: 'rgba(239 | 68 | 68 | 0.8)',
                                        color: '#ffffff',
                                        font: { size: 12, weight: 'bold' }
                                    }
                                }
                            }
                        }
                    }
                }
            });

            // Main update function
            function updateChart() {
                const T = parseFloat(tempSlider.value);
                const M_g = parseFloat(massSlider.value);
                const M_kg = M_g / 1000;

                // Update displayed values
                tempValueSpan.textContent = T;
                massValueSpan.textContent = M_g.toFixed(2);

                // Calculate characteristic speeds
                const vp = Math.sqrt((2 * R * T) / M_kg);
                const vavg = Math.sqrt((8 * R * T) / (Math.PI * M_kg));
                const vrms = Math.sqrt((3 * R * T) / M_kg);

                vpSpan.textContent = vp.toFixed(1);
                vavgSpan.textContent = vavg.toFixed(1);
                vrmsSpan.textContent = vrms.toFixed(1);

                // Generate data for the chart
                const speeds = [];
                const probabilities = [];
                const maxSpeed = Math.max(500, vrms * 3);
                const step = maxSpeed / 250;

                for (let v = 0; v <= maxSpeed; v += step) {
                    speeds.push(v);
                    probabilities.push(maxwellBoltzmann(v, T, M_kg));
                }

                // Update chart data
                mbChart.data.labels = speeds;
                mbChart.data.datasets[0].data = probabilities;

                // Update annotations
                mbChart.options.plugins.annotation.annotations.vpLine.xMin = vp;
                mbChart.options.plugins.annotation.annotations.vpLine.xMax = vp;
                mbChart.options.plugins.annotation.annotations.vavgLine.xMin = vavg;
                mbChart.options.plugins.annotation.annotations.vavgLine.xMax = vavg;
                mbChart.options.plugins.annotation.annotations.vrmsLine.xMin = vrms;
                mbChart.options.plugins.annotation.annotations.vrmsLine.xMax = vrms;

                // Dynamically adjust y-axis max
                const maxProb = Math.max(...probabilities);
                mbChart.options.scales.y.max = maxProb * 1.1;

                mbChart.update();
            }

            // Event listeners for sliders
            tempSlider.addEventListener('input', updateChart);
            massSlider.addEventListener('input', updateChart);

            // Initial chart render
            updateChart();
        });
    </script>
